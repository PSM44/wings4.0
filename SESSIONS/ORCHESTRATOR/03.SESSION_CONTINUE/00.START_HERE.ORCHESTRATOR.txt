### RUNTIME_GATE_POWERSHELL (copy/run verbatim; absolute paths; any cwd)

```powershell
$ErrorActionPreference = 'Stop'
$Root = 'C:\01. GitHub\Wings4.0'
Set-Location -LiteralPath $Root
$Branch = (git branch --show-current).Trim()
$Head = (git rev-parse HEAD).Trim()
$WorktreeDiff = @(git diff --name-only --)
$IndexDiff = @(git diff --cached --name-only --)
$ExpectedAncestors = @(
  '1e8315d3836c941ba8cc6ef4a1b7c6981bb0c33c',
  '16b28d19c6342271b4e5e417a84649a69a14443e',
  '03a589dd3f6b8e7c4d3314c340d550327cb203f3',
  'd23d29294e39287c75140b767bee300a42958a73',
  'e199d8b8296c290d018b4b2b282c9618ceb121f4',
  '6bb3bfcede2fbe0417191bdcf76115fcf1eab483',
  '9e301acfd3d2c633777650d3518d58ed19690826'
)
$AncestorFail = New-Object System.Collections.Generic.List[string]
foreach ($c in $ExpectedAncestors) {
  git merge-base --is-ancestor $c HEAD 2>$null | Out-Null
  if ($LASTEXITCODE -ne 0) { [void]$AncestorFail.Add($c) }
}
$WorktreeClean = ($WorktreeDiff.Count -eq 0).ToString().ToUpperInvariant()
$IndexClean = ($IndexDiff.Count -eq 0).ToString().ToUpperInvariant()
$BaselineIsAncestor = ($AncestorFail.Count -eq 0).ToString().ToUpperInvariant()
$Lines = New-Object System.Collections.Generic.List[string]
[void]$Lines.Add("ROOT=$Root")
[void]$Lines.Add("BRANCH=$Branch")
[void]$Lines.Add("HEAD=$Head")
[void]$Lines.Add("WORKTREE_CLEAN=$WorktreeClean")
[void]$Lines.Add("INDEX_CLEAN=$IndexClean")
[void]$Lines.Add("BASELINE_IS_ANCESTOR=$BaselineIsAncestor")
[void]$Lines.Add("WORKTREE_FILE_COUNT=$($WorktreeDiff.Count)")
[void]$Lines.Add("INDEX_FILE_COUNT=$($IndexDiff.Count)")
if ($AncestorFail.Count -gt 0) { [void]$Lines.Add("ANCESTOR_FAIL=$($AncestorFail -join ';')") }
$Lines | ForEach-Object { $_ }
```
